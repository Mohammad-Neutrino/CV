\section*{Research Experience}

\subsection*{Research Roles and Appointments}
\begin{itemize}
    \item \cvdated{{\bf Postdoctoral Researcher: Particle Astrophysics, University of Kansas}}{Jun 2026--Present}
    Study radio detection of ultra-high energy particles, backgrounds, and detector response with Prof.\ D. Besson.

    \item \cvdated{\textbf{Core Analyst, ARA Five-Station Diffuse Neutrino Search}}{2022--Present}
    One of eight core analysts for the 10.6-year five-station ARA diffuse search. Integrated ARA01 into the common framework and contributed to optimization, unblinding, and interpretation.

    \item \cvdated{\textbf{Analysis Lead, Non-Anthropogenic Down-going Signals in ARA}}{Jul 2026--Present}
    Leading an array-wide search across all five ARA stations for down-going radio signals inconsistent with known anthropogenic backgrounds.

    \item \cvdated{\textbf{Analysis Lead, Radio Background Characterization with ARA}}{Dec 2025--Present}
    Leading studies of astrophysical, environmental, and anthropogenic radio backgrounds in ARA data, including solar activity, galactic noise, high wind triboelectric events, radiosondes, and local sources.

    \item \cvdated{\textbf{Primary Analyzer, Askaryan Radio Array (ARA) Station 1}}{2021--2026}
    Led ARA01 calibration, data-driven detector modeling, reconstruction, background rejection, and simulation. Developed its analysis framework for the array-wide search.

    \item \cvdated{\textbf{Monitoring Shifter, IceCube Neutrino Observatory}}{2021--2026}
    Served as KU representative for monitoring shifts, presenting weekly reports to the collaboration.

    \item \cvdated{\textbf{Visiting Summer Student}}{May--Jul 2019}
    Studied ``Importance of 1-3 Mixing Angle in Three-Flavor Neutrino Oscillations Paradigm'' with Prof.\ S.K. Agarwalla at the Institute of Physics, India, producing technical report   \cvlinkeddoi{https://www.researchgate.net/publication/340095776_Importance_of_1-3_Mixing_Angle_in_Three-Flavor_Neutrino_Oscillations_Paradigm_SSVP_Summer_Project_Report}{10.13140/RG.2.2.34421.12009}.

    \item \cvdated{\textbf{Winter Research Intern}}{Dec 2019--Jan 2020}
    Researched ``Weak Interaction Physics'' at Indian Institute of Technology--BHU with Prof. G. Abbas.
\end{itemize}

\subsection*{Scientific Software and Open-Source Development}
\begin{itemize}
    \item \cvdated{\textbf{MAHAM:} \textbf{M}odules for \textbf{A}stroparticle, \textbf{H}igh-energy, \textbf{A}stronomy and \textbf{M}ultimessenger}{2026--Present}
    Founder and lead developer of \href{https://github.com/Project-MAHAM}{\tt Project MAHAM}, an open-source framework for validated and reusable methods in astroparticle physics and multimessenger astronomy. Designed its architecture and provenance checks; developing neutrino spectra, experimental limits, detector response, statistics, and literature datasets.

    \item \cvdated{\textbf{AAFIYA:} \textbf{A}ntenna \textbf{A}nalysis in \textbf{F}requency-domain for \textbf{I}mpedance \& \textbf{Y}ield \textbf{A}ssessment}{2024--Present}
    Creator of \href{https://github.com/Mohammad-Neutrino/AAFIYA}{\tt AAFIYA}, an open-source toolkit for antenna measurements and simulations, including S-parameters, impedance, gain patterns, polarization, vector effective length, and measurement-to-simulation comparisons.

    \item \cvdated{\textbf{AraProc: ARA Data Analysis Framework}}{2024--2026}
    Co-developer of the ARA Collaboration's
    \href{https://github.com/ara-software/AraProc}{\tt AraProc} Python analysis framework, with 21 merged pull requests covering waveform analysis, event reconstruction, CW filtering, and data quality improvements.
\end{itemize}
